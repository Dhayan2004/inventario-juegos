## Ejemplo 4 — Modal.tsx (confirm variant)

```typescript
'use client'  // requiere state (open) + portal + escape handling

const modalContentVariants = cva(
  'fixed left-1/2 top-1/2 z-50 -translate-x-1/2 -translate-y-1/2 rounded-md bg-surface-elevated border border-border duration-base data-[state=open]:animate-in data-[state=closed]:animate-out',
  {
    variants: {
      variant: {
        confirm: 'p-6 max-w-md w-full',
        form:    'p-6 max-w-2xl w-full',
        detail:  'p-6 max-w-3xl w-full max-h-[80vh] overflow-y-auto',
      },
    },
    defaultVariants: { variant: 'confirm' },
  }
)

// Modal.confirm pattern
export function ConfirmModal({
  title,
  description,
  confirmLabel = 'Confirmar',
  cancelLabel = 'Cancelar',
  destructive = false,
  onConfirm,
  onCancel,
  children,
}: ConfirmModalProps) {
  return (
    <Dialog>
      <DialogOverlay className="fixed inset-0 z-40 bg-black/80 data-[state=open]:animate-in data-[state=closed]:animate-out" />
      <DialogContent className={cn(modalContentVariants({ variant: 'confirm' }))} role="dialog" aria-modal="true">
        <h2 className="font-display text-lg font-semibold text-text">{title}</h2>
        {description && <p className="mt-2 text-sm text-text-muted">{description}</p>}
        {children}
        <div className="mt-6 flex justify-end gap-2">
          <Button variant="secondary" onClick={onCancel}>{cancelLabel}</Button>
          <Button variant={destructive ? 'danger' : 'primary'} onClick={onConfirm}>{confirmLabel}</Button>
        </div>
      </DialogContent>
    </Dialog>
  )
}
```

**Brand Score:** 90/100.
**Rules enforced:**
- Contextual title required → `title` prop required
- Escape/cancel behavior → handled by Dialog primitive (or manual onKeyDown if from-scratch)
- Destructive actions require explicit confirmation → `destructive` flag changes button.danger
- Focus trap → handled by Dialog primitive

## Ejemplo 6 — DataTable.tsx (Mode B derivation)

Este es Mode B. NO está en `component_rules`. Derivation: `card.metric + form.search`.

```typescript
'use client'  // sort state

/**
 * DataTable · derived component (NOT in brand.json.component_rules)
 *
 * Purpose: muestra rows comparables y permite reordenar para encontrar outliers
 *
 * Derivation rationale (R-005 sec 8.2):
 *   nearest_component: card.metric + form.search
 *   tokens applied:    surface-elevated, border, font-display headers,
 *                      font-mono amounts, focus-ring=primary
 *   posture-shaped:    density=4 → compact padding + sticky header
 *                      geometry=4 → radius-md (no lg)
 *                      materiality=1 → no shadow
 *   inherited rules:   focus-visible row (card.metric), label clear (form pattern)
 *   additional rules:  aria-sort en sortable headers, keyboard row nav
 *
 * must_include:
 *   states:        focus-visible row, hover row, active selected, disabled
 *   responsive:    mobile = horizontal scroll + sticky first col
 *   accessibility: role=table, th scope=col, aria-sort, keyboard rows
 */

interface Column<T> {
  key: keyof T
  label: string
  sortable?: boolean
  align?: 'left' | 'right' | 'center'
  format?: (value: T[keyof T]) => React.ReactNode
}

interface DataTableProps<T> {
  data: T[]
  columns: Column<T>[]
  onRowSelect?: (row: T) => void
  emptyState?: React.ReactNode
}

export function DataTable<T extends { id: string | number }>({ data, columns, onRowSelect, emptyState }: DataTableProps<T>) {
  const [sortKey, setSortKey] = useState<keyof T | null>(null)
  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')

  // ... sort logic + keyboard nav

  if (data.length === 0) {
    return (
      <Card variant="empty">
        {emptyState ?? <p>No hay datos para mostrar.</p>}
      </Card>
    )
  }

  return (
    <Card variant="default" className="overflow-x-auto">
      <table role="table" className="w-full text-sm">
        <thead className="sticky top-0 bg-surface-elevated">
          <tr>
            {columns.map((col) => (
              <th
                key={String(col.key)}
                scope="col"
                aria-sort={sortKey === col.key ? (sortDir === 'asc' ? 'ascending' : 'descending') : 'none'}
                className="font-mono text-xs uppercase tracking-widest text-text-subtle px-3 py-2 text-left"
              >
                {col.sortable ? (
                  <button onClick={() => toggleSort(col.key)} className="hover:text-text">
                    {col.label}
                  </button>
                ) : col.label}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {sortedData.map((row) => (
            <tr
              key={row.id}
              tabIndex={0}
              onClick={() => onRowSelect?.(row)}
              className="border-t border-border hover:bg-surface-higher focus-visible:outline-none focus-visible:bg-surface-higher cursor-pointer"
            >
              {columns.map((col) => (
                <td key={String(col.key)} className="px-3 py-2 text-text">
                  {col.format ? col.format(row[col.key]) : String(row[col.key])}
                </td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </Card>
  )
}
```

## Patterns observados

1. **CSS vars via tailwind extend** — todos los componentes usan `bg-primary`, `text-text-muted`, `font-display` (clases sugar derivadas en `tailwind.config.ts`) en vez de `bg-[var(--color-primary)]`. Posible porque referenciamos `tailwind-token-bridge.md` extends.
2. **cva siempre** — base class string + variants object. defaultVariants explícito.
3. **`'use client'` solo cuando hace falta** — Modal, Tabs, DataTable (state). Button, Card son Server Components.
4. **forwardRef en componentes interactivos** — Button, Input. NO en presentational (Card.empty).
5. **Composition over kitchen-sink** — DataTable usa Card + table semantics nativos en vez de un mega-component.
6. **JSDoc rico** — citations, rules enforced, must_include para Mode B.
